35template <typename I2cType>
36bool pca9685::PCA9685<I2cType>::Reset() noexcept {
37 // Ensure I2C bus is initialized and ready (mirrors PCAL95555 pattern)
38 if (!i2c_ || !i2c_->EnsureInitialized()) {
39 setError(Error::I2cWrite);
40 initialized_ = false;
41 return false;
42 }
43
44 // MODE1 = AI only: oscillator on (SLEEP = 0), register auto-increment on (datasheet 7.3.1,
45 // Table 5) so that the four-byte LEDn / ALL_LED frames reach all four registers.
46 if (!writeMode1(0x00)) {
47 initialized_ = false;
48 return false;
49 }
50 waitOscillatorStartup(); // SLEEP may have been 1 (it is at power-up): 500 us start-up
51 initialized_ = true;
52 last_error_ = Error::None;
53 return true;
54}

56template <typename I2cType>
57bool pca9685::PCA9685<I2cType>::SetPwmFreq(float freq_hz) noexcept {
58 if (!EnsureInitialized()) {
59 setError(Error::NotInitialized);
60 return false;
61 }
62 if (freq_hz < 24.0F || freq_hz > 1526.0F) {
63 setError(Error::OutOfRange);
64 return false;
65 }
66 uint8_t prescale = calcPrescale(freq_hz);
67 uint8_t old_mode = 0;
68 if (!readReg(static_cast<uint8_t>(Register::MODE1), old_mode)) {
69 return false;
70 }
71 // PRE_SCALE accepts writes only while MODE1.SLEEP = 1 (Table 4, note 1)
72 if (!writeMode1(static_cast<uint8_t>(old_mode | MODE1_SLEEP_))) {
73 return false;
74 }
75 if (!writeReg(static_cast<uint8_t>(Register::PRE_SCALE), prescale)) {
76 return false;
77 }
78 if ((old_mode & MODE1_SLEEP_) == 0) {
79 // Was running: wake, give the oscillator 500 us, then write RESTART so that the channels
80 // that were running resume (datasheet 7.3.1.1 steps 2 and 3). A 1 written to RESTART when it
81 // reads 0 has no effect (Table 5), so it is written unconditionally.
82 if (!restartAfterSleep(old_mode, true)) {
83 return false;
84 }
85 }
86 last_error_ = Error::None;
87 return true;
88}

92 uint16_t off_time) noexcept {

94 setError(Error::NotInitialized);
95 return false;

113bool pca9685::PCA9685<I2cType>::SetDuty(uint8_t channel, float duty) noexcept {
114 duty = ::std::max(duty, 0.0F);
115 duty = ::std::min(duty, 1.0F);
116 if (duty >= 1.0F) {
117 // 100 %: the full-ON bit, LEDn_ON_H[4] (datasheet 7.3.3, Table 7, p. 24). ON = 0, OFF = 4095
118 // would leave the output low for one tick of every period.
119 return SetChannelFullOn(channel);
120 }
121 // Rounds to 0 for duty < 1/8190: SetPwm() turns ON = OFF = 0 into full OFF.
122 auto off_time = static_cast<uint16_t>(lroundf((duty * MAX_PWM_)));
123 return SetPwm(channel, 0, off_time);
124}

176template <typename I2cType>
177bool pca9685::PCA9685<I2cType>::Wake() noexcept {
178 if (!EnsureInitialized()) {
179 setError(Error::NotInitialized);
180 return false;
181 }
182 uint8_t mode1 = 0;
183 if (!readReg(static_cast<uint8_t>(Register::MODE1), mode1)) {
184 return false;
185 }
186 const bool asleep = (mode1 & MODE1_SLEEP_) != 0;
187 const bool restart = (mode1 & MODE1_RESTART_) != 0;
188 if (asleep || restart) {
189 // Datasheet 7.3.1.1: clear SLEEP, allow 500 us for the oscillator, then write 1 to RESTART
190 // if it read 1, so that the channels that were running resume.
191 if (!restartAfterSleep(mode1, restart)) {
192 return false;
193 }
194 }
195 last_error_ = Error::None;

345template <typename I2cType>
346bool pca9685::PCA9685<I2cType>::writeMode1(uint8_t value) noexcept {
347 // Keep AI set on every MODE1 write, and never write 1 to RESTART by accident: a 1 there restarts
348 // the PWM channels (datasheet 7.3.1, Table 5 and 7.3.1.1).
349 const auto mode1 = static_cast<uint8_t>((value & ~MODE1_RESTART_) | MODE1_AI_);
350 return writeReg(static_cast<uint8_t>(Register::MODE1), mode1);
351}

353template <typename I2cType>
354bool pca9685::PCA9685<I2cType>::restartAfterSleep(uint8_t mode1, bool restart) noexcept {
355 const auto awake = static_cast<uint8_t>(mode1 & ~MODE1_SLEEP_);
356 if (!writeMode1(awake)) {
357 return false;
358 }
359 waitOscillatorStartup(); // SLEEP must be 0 for 500 us before RESTART is written (7.3.1.1)
360 if (!restart) {
361 return true;
362 }
363 return writeReg(static_cast<uint8_t>(Register::MODE1),
364 static_cast<uint8_t>((awake & ~MODE1_RESTART_) | MODE1_AI_ | MODE1_RESTART_));
365}

367template <typename I2cType>
368void pca9685::PCA9685<I2cType>::waitOscillatorStartup() noexcept {
369 if constexpr (requires(I2cType& bus, uint32_t us) { bus.DelayUs(us); }) {
370 i2c_->DelayUs(OSC_STARTUP_US_);
371 } else {
372 // The port has no DelayUs(): time the wait with bus traffic instead. A one-byte register read
373 // is at least 39 SCL periods (START, 4 x 9 bits, repeated START, STOP), 39 us at the chip's
374 // fastest clock of 1 MHz (Table 15), so 13 reads of MODE1 take at least 507 us. MODE1 is not a
375 // PWM control register, so reading it inside the start-up window is allowed (Table 5, note 2).
376 uint8_t scratch = 0;
377 for (uint32_t i = 0; i < OSC_STARTUP_FALLBACK_READS_; ++i) {
378 (void)i2c_->Read(addr_, static_cast<uint8_t>(Register::MODE1), &scratch, 1);
379 }
380 }
381}

383template <typename I2cType>
384::std::array<uint8_t, 4> pca9685::PCA9685<I2cType>::pwmFrame(uint16_t on_time,
385 uint16_t off_time) noexcept {
386 if (on_time == off_time) {
387 // LEDn_ON and LEDn_OFF must never hold the same count (datasheet 7.3.4, p. 25). Equal ticks
388 // mean zero duty, so write the full-OFF bit, LEDn_OFF_H[4], instead.
389 return {0x00, 0x00, 0x00, 0x10};
390 }
391 return {static_cast<uint8_t>(on_time & 0xFF), static_cast<uint8_t>((on_time >> 8) & 0x0F),
392 static_cast<uint8_t>(off_time & 0xFF), static_cast<uint8_t>((off_time >> 8) & 0x0F)};
393}

397 uint8_t current = 0;
398 if (!readReg(reg, current)) {
399 return false;

401 uint8_t new_val = (current & static_cast<uint8_t>(~mask)) | (value & mask);

409template <typename I2cType>
410uint8_t pca9685::PCA9685<I2cType>::calcPrescale(float freq_hz) const noexcept {
411 float prescale_val = (static_cast<float>(OSC_FREQ_) / (4096.0F * freq_hz)) - 1.0F;
412 prescale_val = ::std::max(prescale_val, 3.0F);
413 prescale_val = ::std::min(prescale_val, 255.0F);
414 return static_cast<uint8_t>(lroundf(prescale_val));
415}
