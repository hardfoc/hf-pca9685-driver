35template <typename I2cType>
36bool pca9685::PCA9685<I2cType>::Reset() noexcept {
37 // Ensure I2C bus is initialized and ready (mirrors PCAL95555 pattern)
38 if (!i2c_ || !i2c_->EnsureInitialized()) {
39 setError(Error::I2cWrite);
40 initialized_ = false;
41 return false;
42 }
43
44 uint8_t mode1 = 0x00; // Reset value
45 if (!writeReg(static_cast<uint8_t>(Register::MODE1), mode1)) {
46 initialized_ = false;
47 return false;
48 }
49 initialized_ = true;
50 last_error_ = Error::None;
51 return true;
52}

54template <typename I2cType>
55bool pca9685::PCA9685<I2cType>::SetPwmFreq(float freq_hz) noexcept {
56 if (!EnsureInitialized()) {
57 setError(Error::NotInitialized);
58 return false;
59 }
60 if (freq_hz < 24.0F || freq_hz > 1526.0F) {
61 setError(Error::OutOfRange);
62 return false;
63 }
64 uint8_t prescale = calcPrescale(freq_hz);
65 uint8_t old_mode = 0;
66 if (!readReg(static_cast<uint8_t>(Register::MODE1), old_mode)) {
67 return false;
68 }
69 uint8_t sleep = (old_mode & 0x7F) | 0x10; // sleep
70 if (!writeReg(static_cast<uint8_t>(Register::MODE1), sleep)) {
71 return false;
72 }
73 if (!writeReg(static_cast<uint8_t>(Register::PRE_SCALE), prescale)) {
74 return false;
75 }
76 if (!writeReg(static_cast<uint8_t>(Register::MODE1), old_mode)) {
77 return false;
78 }
79 last_error_ = Error::None;
80 return true;
81}

83template <typename I2cType>
84bool pca9685::PCA9685<I2cType>::SetPwm(uint8_t channel, uint16_t on_time,
85 uint16_t off_time) noexcept {
86 if (!EnsureInitialized()) {
87 setError(Error::NotInitialized);
88 return false;
89 }
90 if (channel >= MAX_CHANNELS_ || on_time > MAX_PWM_ || off_time > MAX_PWM_) {
91 setError(Error::OutOfRange);
92 return false;
93 }
94 uint8_t reg = static_cast<uint8_t>(Register::LED0_ON_L) + (4 * channel);
95 ::std::array<uint8_t, 4> data = {
96 static_cast<uint8_t>(on_time & 0xFF), static_cast<uint8_t>((on_time >> 8) & 0x0F),
97 static_cast<uint8_t>(off_time & 0xFF), static_cast<uint8_t>((off_time >> 8) & 0x0F)};
98 if (!writeRegBlock(reg, data.data(), 4)) {
99 return false;
100 }
101 last_error_ = Error::None;
102 return true;
103}

159template <typename I2cType>
160bool pca9685::PCA9685<I2cType>::Wake() noexcept {
161 if (!EnsureInitialized()) {
162 setError(Error::NotInitialized);
163 return false;
164 }
165 uint8_t mode1 = 0;
166 if (!readReg(static_cast<uint8_t>(Register::MODE1), mode1)) {
167 return false;
168 }
169 uint8_t new_mode1 = mode1 & static_cast<uint8_t>(~0x10U); // Clear SLEEP
170 if (!writeReg(static_cast<uint8_t>(Register::MODE1), new_mode1)) {
171 return false;
172 }
173 // If RESTART bit was set, set it again to restart PWM channels
174 if ((mode1 & 0x80U) != 0) {
175 // Oscillator needs ~500us to stabilize; caller should delay if needed
176 if (!writeReg(static_cast<uint8_t>(Register::MODE1), static_cast<uint8_t>(new_mode1 | 0x80U))) {
177 return false;
178 }
179 }
180 last_error_ = Error::None;
181 return true;
182}

319template <typename I2cType>
320bool pca9685::PCA9685<I2cType>::modifyReg(uint8_t reg, uint8_t mask, uint8_t value) noexcept {
321 uint8_t current = 0;
322 if (!readReg(reg, current)) {
323 return false;
324 }
325 uint8_t new_val = (current & static_cast<uint8_t>(~mask)) | (value & mask);
326 if (!writeReg(reg, new_val)) {
327 return false;
328 }
329 last_error_ = Error::None;
330 return true;
331}

333template <typename I2cType>
334uint8_t pca9685::PCA9685<I2cType>::calcPrescale(float freq_hz) const noexcept {
335 float prescale_val = (static_cast<float>(OSC_FREQ_) / (4096.0F * freq_hz)) - 1.0F;
336 prescale_val = ::std::max(prescale_val, 3.0F);
337 prescale_val = ::std::min(prescale_val, 255.0F);
338 return static_cast<uint8_t>(lroundf(prescale_val));
339}
